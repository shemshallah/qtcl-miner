SERVER SENDS → CLIENT RECEIVES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

HTTP 200 OK
Content-Type: application/json

{
    "jsonrpc": "2.0",
    "result": {
        "status": "healthy",
        "uptime_seconds": 86400
    },
    "id": 1
}

OR (on error):

HTTP 200 OK
Content-Type: application/json

{
    "jsonrpc": "2.0",
    "error": {
        "code": -32601,
        "message": "Method not found"
    },
    "id": 1
}

CRITICAL RULES (DO NOT BREAK THESE)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

✓ ALWAYS return HTTP 200 (never 503, 404, 500, etc)
✓ MUST include "jsonrpc": "2.0" in response
✓ MUST include "id" that matches the request
✓ MUST have EITHER "result" OR "error", never both
✓ Set Content-Type: application/json header
✓ Invalid method = JSON error response, not HTTP error
✓ Server crash = JSON error response, not HTTP error

ERROR CODES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

-32700  Parse error           Invalid JSON
-32600  Invalid Request       Wrong format
-32601  Method not found      Unknown method
-32602  Invalid params        Wrong param count/type
-32603  Internal error        Server crash
-32000  Application error     Custom error

COMMON MISTAKES TO AVOID
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

❌ Missing "jsonrpc": "2.0" field
❌ Missing "id" field in response
❌ Having both "result" and "error" in response
❌ Not setting Content-Type: application/json header
❌ Returning HTML error pages instead of JSON
❌ Using GET instead of POST
❌ Wrong endpoint path (/api/rpc instead of /rpc)
❌ Putting method in URL instead of JSON body
❌ Not validating JSON-RPC format before processing
